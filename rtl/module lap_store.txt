module lap_store (
    input  wire        CLK,
    input  wire        timer_reset,
    input  wire        timer_lap,

    input  wire [3:0] m10,
    input  wire [3:0] m1,
    input  wire [3:0] s10,
    input  wire [3:0] s1,
    input  wire [3:0] ms10,
    input  wire [3:0] ms1,

    output reg  [3:0] lap_m10  = 4'b0000,
    output reg  [3:0] lap_m1   = 4'b0000,
    output reg  [3:0] lap_s10  = 4'b0000,
    output reg  [3:0] lap_s1   = 4'b0000,
    output reg  [3:0] lap_ms10 = 4'b0000,
    output reg  [3:0] lap_ms1  = 4'b0000
   
);


always @(posedge CLK) begin
    if (timer_reset) begin
       
        lap_m10  <= 4'b0000;
        lap_m1   <= 4'b0000;
        lap_s10  <= 4'b0000;
        lap_s1   <= 4'b0000;
        lap_ms10 <= 4'b0000;
        lap_ms1  <= 4'b0000;
    end
    else begin
        if (timer_lap) begin
          
            lap_m10  <= m10;
            lap_m1   <= m1;
            lap_s10  <= s10;
            lap_s1   <= s1;
            lap_ms10 <= ms10;
            lap_ms1  <= ms1;
        end 
    end     
	 end        

endmodule
